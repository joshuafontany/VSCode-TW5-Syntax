# Where TiddlyWiki builds a construct and the grammar answers with nothing of its own.
#
# tools/darkness-witness.js reads every wikitext carrier under both readers and judges each line
# of each construct the host builds. A line here names darkness already looked at; any other dark
# line fails the gate, and a line here that reads lit fails as stale — so a repair retires the debt
# it paid, and a declaration cannot outlive what it explains.
#
#   <carrier>  <rule>  <MISS|LOST>  "<the line, trimmed>"  # reason
#
# MISS  the grammar reads nothing of the line even with the construct standing alone — no rule
# LOST  it reads the construct alone and loses the line in place — text above holds a region
#
# The key carries the line's TEXT rather than its number, so an edit elsewhere in a carrier moves
# nothing here, and an edit to the line itself asks for the judgement again.
#
# OWED marks a divergence standing as debt. A line reading dark on purpose carries its ruling
# instead, and says why a reader should meet plain text where the host builds markup.
#
# `node tools/darkness-witness.js --list` prints every dark line in this form, keys derived.

# A LIST ITEM AFTER A LINE CLOSING AN HTML BLOCK. TiddlyWiki keeps the list open past `</div> run`
# and opens a fresh item on the next `*`. The grammar has fallen into a paragraph by then, so the
# marker reads as prose — and reads so with the whole list standing alone.
tests/samples/tiddlywiki5.basic.tw  list  MISS  "* Inline <span>"  # OWED — the host opens a fresh item here; the grammar reads a paragraph
tests/samples/tiddlywiki5.basic.tw  list  MISS  "* <$my-widget"  # OWED — the host opens a fresh item here; the grammar reads a paragraph
tests/samples/tiddlywiki5.inline.widgets.tw  list  MISS  "* Inline <span>"  # OWED — the host opens a fresh item here; the grammar reads a paragraph
tests/samples/tiddlywiki5.inline.widgets.tw  list  MISS  "* <$my-widget"  # OWED — the host opens a fresh item here; the grammar reads a paragraph
tests/samples/tiddlywiki5.tw  list  MISS  "* Inline <span>"  # OWED — the host opens a fresh item here; the grammar reads a paragraph
tests/samples/tiddlywiki5.tw  list  MISS  "* <$my-widget"  # OWED — the host opens a fresh item here; the grammar reads a paragraph

# A WIDGET'S ATTRIBUTES ACROSS LINES. The widget opens at line 81 and TiddlyWiki reads every line
# below as its own — backtick-substituted values spanning lines among them. The grammar reads none
# of these lines, even with the widget standing alone.
tests/samples/tiddlywiki5.inline.widgets.tw  html  MISS  "string-replaced-attribute=`"  # OWED — a backtick-substituted value opening across lines
tests/samples/tiddlywiki5.inline.widgets.tw  html  MISS  "`"  # OWED — the backtick closing that value on a later line
tests/samples/tiddlywiki5.inline.widgets.tw  html  MISS  "string-replaced-block=```"  # OWED — a triple-backtick value opening across lines
tests/samples/tiddlywiki5.inline.widgets.tw  html  MISS  "paragaph continues, then ends"  # OWED — the interior of that value
tests/samples/tiddlywiki5.inline.widgets.tw  html  MISS  "```"  # OWED — the triple backtick closing it
tests/samples/tiddlywiki5.inline.widgets.tw  html  MISS  "fitler=\"[filter[test]]\" #broken"  # OWED — an attribute the host still reads as the widget's own
corpus/wikitext/html.multiline-tags.tw  html  MISS  "name=\"targetTiddler\""  # OWED — an attribute on a later line of a multi-line tag

# ITALIC ACROSS LINES. TiddlyWiki carries italic across a line inside one block. The grammar's
# italic opens only where its close stands on the same line — `//(?=[^\n]*//)` — the bound that
# keeps an unclosed `//` from running past its own line.
corpus/wikitext/inline.emphasis.tw  italic  MISS  "//italic across"  # OWED — italic the host carries across a line
corpus/wikitext/inline.emphasis.tw  italic  MISS  "two lines//"  # OWED — italic the host carries across a line

# A PRAGMA BELOW CONTENT. TiddlyWiki reads pragmas only above a tiddler's content, so this
# `\procedure` stands as prose, and the host builds a multi-valued display from the `((…))` inside it.
# The grammar reads a procedure signature there and nothing of the display.
corpus/wikitext/spans.multiline.tw  mvvdisplayinline  MISS  "]] d:((a variable default"  # OWED — a display the host builds inside prose the grammar reads as a signature
corpus/wikitext/spans.multiline.tw  mvvdisplayinline  MISS  ")))"  # OWED — that display's close

# CONSTRUCTS A REGION ABOVE HOLDS. Each reads as itself alone and as something else in place.
corpus/wikitext/degenerate.unterminated.tw  html  LOST  "<div class=\"never-closed\">"  # OWED — the host builds this element; a region opened above it holds the grammar here
corpus/wikitext/inline.links.tw  extlink  LOST  "which is the reading a reader sees: [ext[https://example.com/unclosed"  # OWED — the host builds this external link; the grammar reads it alone and loses it in place
tests/samples/tiddlywiki5.lists.tw  codeinline  LOST  "* `raw``raw`not raw"  # OWED — the host builds two code runs side by side; the grammar reads each alone and loses them together
tests/samples/tiddlywiki5.tw  codeinline  LOST  "* `raw``raw`not raw"  # OWED — the host builds two code runs side by side; the grammar reads each alone and loses them together

# WHITESPACE BEFORE A BLOCK. TiddlyWiki skips leading whitespace, tabs included, before a block rule,
# so `!!!` behind tabs still opens a heading. Read alone behind spaces, the grammar reads the heading.
tests/samples/test.html.tw  heading  LOST  "!!!\t\t//content// {{transclude!!field}}"  # OWED — a heading behind tabs the host builds and the grammar loses
